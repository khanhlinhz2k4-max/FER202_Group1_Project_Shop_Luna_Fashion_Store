\chapter{Tổng quan Hệ thống (System Overview)}

\section{Giới thiệu dự án}
\textbf{LUNE Fashion Store} là nền tảng thương mại điện tử chuyên biệt cho ngành hàng thời trang may mặc và phụ kiện cao cấp dành cho cả phái đẹp và phái mạnh (\textit{Unisex Luxury Fashion Brand}). Hệ thống được xây dựng theo triết lý thiết kế \textbf{Warm Luxury / Quiet Elegance}, mang lại trải nghiệm mua sắm trực quan, liền mạch cho khách hàng, đồng thời cung cấp cổng quản trị vận hành (\textit{Admin Portal}) toàn diện cho quản trị viên.

\section{Mục tiêu hệ thống}
\begin{itemize}[leftmargin=1.5cm]
    \item \textbf{Về phía người dùng (Storefront):} Cung cấp trải nghiệm tìm kiếm trực tiếp theo thời gian thực (\textit{Live Search}), bộ lọc sản phẩm đa tiêu chí, ngăn kéo giỏ hàng tương tác nhanh (\textit{Slide-in Cart Drawer}), quy trình đặt hàng tích hợp mã giảm giá và VietQR, theo dõi tiến trình đơn hàng 4 giai đoạn, xuất hóa đơn in ấn và trợ lý tư vấn kích cỡ thông minh 24/7 (\textit{AI Sizing Chatbot}).
    \item \textbf{Về phía quản trị (Admin Portal):} Cung cấp bảng điều khiển số liệu kinh doanh trực quan (\textit{KPI Dashboard} kèm biểu đồ SVG), quản lý vòng đời sản phẩm (\textit{CRUD} \& ẩn/hiện 1-click), quản lý tồn kho tự động, điều phối trạng thái đơn hàng, quản lý chương trình khuyến mãi (\textit{Voucher}) và danh sách người dùng.
    \item \textbf{Về mặt kỹ thuật:} Ứng dụng kiến trúc Single Page Application (SPA) với React 19, React Router 7, Context API, tích hợp thư viện thành phần \textbf{shadcn/ui} chuẩn hóa và hệ thống đa ngôn ngữ Song ngữ Việt -- Anh 100\%.
\end{itemize}

\chapter{Phân tích Yêu cầu \& Kiến trúc Hệ thống}

\section{Các tác nhân hệ thống (System Actors)}
Hệ thống phân quyền thành 3 tác nhân chính với phạm vi truy cập rõ ràng:

\subsection{Khách vãng lai (Guest)}
\begin{itemize}[leftmargin=1.5cm]
    \item Duyệt toàn bộ giao diện công khai: Trang chủ (\texttt{/}), Cửa hàng (\texttt{/shop}), Chi tiết sản phẩm (\texttt{/product/:id}).
    \item Sử dụng thanh tìm kiếm trực tiếp (\textit{Live Search}), bộ lọc đa tiêu chí, sắp xếp sản phẩm, cửa sổ Xem nhanh (\textit{Quick View}) và Bảng hướng dẫn kích cỡ (\textit{Size Guide}).
    \item Thêm sản phẩm vào giỏ hàng, lưu danh sách yêu thích tạm thời (\textit{Guest Lookbook}) và đặt hàng tại trang Thanh toán (\texttt{/checkout}).
    \item Trò chuyện với Trợ lý Chatbot 24/7 để nhận tư vấn chọn size theo chiều cao, cân nặng và số đo 3 vòng.
\end{itemize}

\subsection{Khách hàng thành viên (Registered Customer -- \texttt{role: user})}
\begin{itemize}[leftmargin=1.5cm]
    \item Kế thừa toàn bộ quyền hạn của Khách vãng lai và tự động đồng bộ danh sách Yêu thích cá nhân (\textit{Auto-merge Wishlist}).
    \item Quản lý không gian cá nhân 5 Tab tại \texttt{/profile}: Lịch sử đơn hàng (\textit{My Orders}), Thông tin cá nhân (\textit{Personal Details}), Bảo mật \& Đổi mật khẩu (\textit{Password \& Security}), Sổ địa chỉ (\textit{Address Book}) và Danh sách yêu thích (\textit{Saved Wishlist}).
    \item Theo dõi tiến trình đơn hàng qua thanh \texttt{OrderTimeline} 4 nấc, in hóa đơn chuẩn Boutique (\texttt{InvoiceModal}) và gửi đánh giá sao xác thực (\textit{Verified Buyer}) khi đơn hàng đã giao thành công (\texttt{Delivered}).
\end{itemize}

\subsection{Quản trị viên (Administrator -- \texttt{role: admin})}
\begin{itemize}[leftmargin=1.5cm]
    \item Toàn quyền truy cập cổng quản trị tại \texttt{/admin} (được bảo vệ bởi \texttt{ProtectedRoute}).
    \item \textbf{Dashboard (\texttt{/admin}):} Giám sát 4 chỉ số KPI (Tổng doanh thu, Tổng đơn hàng, Tổng sản phẩm, Tổng thành viên), biểu đồ xu hướng doanh thu 7 ngày và biểu đồ tròn phân bổ danh mục.
    \item \textbf{Quản lý sản phẩm (\texttt{/admin/products}):} Thêm, sửa, xóa sản phẩm, cập nhật tồn kho, bật/tắt hiển thị sản phẩm 1-click (\texttt{Eye / EyeOff}) và xem trước sản phẩm ẩn (\textit{Admin Preview Mode}).
    \item \textbf{Quản lý đơn hàng (\texttt{/admin/orders}):} Duyệt và chuyển trạng thái đơn hàng (\texttt{Pending} $\rightarrow$ \texttt{Shipping} $\rightarrow$ \texttt{Delivered} hoặc \texttt{Cancelled} kèm cơ chế tự động hoàn kho).
    \item \textbf{Quản lý mã giảm giá (\texttt{/admin/vouchers}):} Tạo, sửa, bật/tắt và xóa các mã ưu đãi theo phần trăm hoặc số tiền cố định.
    \item \textbf{Quản lý người dùng (\texttt{/admin/users}):} Tra cứu danh sách tài khoản thành viên và vai trò trên hệ thống.
\end{itemize}

\section{Kiến trúc Dữ liệu \& Lưu trữ (Data Contract)}
Hệ thống chuẩn hóa các mô hình dữ liệu và đồng bộ hóa qua \texttt{localStorage}:
\begin{itemize}[leftmargin=1.5cm]
    \item \texttt{lune\_products}: Lưu danh sách sản phẩm gồm \texttt{id, name, price, category, stock, isActive, image, images, sizes, colors, description, isNew}.
    \item \texttt{lune\_cart}: Lưu giỏ hàng hiện tại gồm \texttt{id, name, price, image, selectedSize, selectedColor, quantity}.
    \item \texttt{lune\_wishlist} \& \texttt{lune\_wishlist\_<email>}: Lưu danh sách yêu thích của khách và đồng bộ theo từng tài khoản thành viên.
    \item \texttt{lune\_orders}: Lưu lịch sử đơn hàng gồm \texttt{id, customer, items, totalAmount, voucherCode, status, createdAt}.
    \item \texttt{lune\_vouchers}: Lưu danh sách mã ưu đãi gồm \texttt{id, code, discountType, discountValue, minSubtotal, startsAt, expiresAt, isActive}.
    \item \texttt{lune\_users} \& \texttt{lune\_current\_user}: Lưu danh sách người dùng và phiên đăng nhập hiện hành.
    \item \texttt{lune\_language}: Lưu tùy chọn ngôn ngữ giao diện (\texttt{vi} hoặc \texttt{en}).
\end{itemize}

\section{Hệ thống Typography \& Thang đo chữ (Modular Type Scale)}
\begin{itemize}[leftmargin=1.5cm]
    \item \textbf{Phông chữ Serif tiêu đề (\texttt{--font-serif}):} \texttt{'Playfair Display', 'Cormorant Garamond', serif} -- mang phong cách biên tập tạp chí thời trang cao cấp, áp dụng cho tiêu đề \texttt{h1--h4}, logo \texttt{LUNE}, tiêu đề Modal và hóa đơn.
    \item \textbf{Phông chữ Sans-serif nội dung (\texttt{--font-sans}):} \texttt{'Montserrat', sans-serif} -- hiện đại, sắc nét, áp dụng cho văn bản mô tả, nút bấm, bảng biểu Admin và các chỉ số tài chính (\texttt{lining-nums tabular-nums}).
    \item \textbf{Thang đo Modular 9 cấp:} Chuẩn hóa từ \texttt{--text-hero (3.5rem)} xuống \texttt{--text-h1 (2.4rem)}, \texttt{--text-h2 (2.0rem)}, \texttt{--text-body (0.95rem)} đến \texttt{--text-caption (0.7rem)}, đảm bảo sự đồng bộ trên mọi độ phân giải màn hình.
\end{itemize}

\section{Các điểm nhấn trải nghiệm UI/UX nâng cao}
\begin{enumerate}[leftmargin=1.5cm]
    \item \textbf{Kiến trúc Component chuẩn shadcn/ui (\texttt{src/components/ui/}):} Tích hợp đầy đủ \texttt{Button}, \texttt{Badge}, \texttt{Dialog}, \texttt{Avatar} dựa trên Radix UI và \texttt{class-variance-authority}, kết hợp cấu hình Tailwind \texttt{preflight: false} để hoạt động mượt mà với hệ thống Vanilla CSS.
    \item \textbf{Trợ lý Tư vấn Thời trang 24/7 (\texttt{ChatBot.jsx}):} Tự động phân tích chiều cao (ví dụ \texttt{1m60}), cân nặng (\texttt{52kg}) và số đo 3 vòng (\texttt{86-66-90}) từ câu chat tự nhiên để đề xuất size may đo (\texttt{XS--XL}), kèm thanh gợi ý nhanh (\textit{Quick Chips}) và hỗ trợ song ngữ Việt -- Anh.
    \item \textbf{Hệ thống Song ngữ Toàn diện (100\% Bilingual VI/EN):} Từ điển hơn 1,500 dòng tại \texttt{src/data/translations.js} giúp chuyển đổi tức thời không cần tải lại trang trên cả Storefront lẫn Admin Portal.
    \item \textbf{Tương tác chiều sâu 3D \& Hóa đơn in ấn Boutique:} Hiệu ứng nâng thẻ sản phẩm 3D tinh tế, con trỏ bụi vàng lấp lánh (\texttt{GoldenCursorTrail.jsx}) tự động tắt trên thiết bị cảm ứng, và chế độ in hóa đơn chuyên biệt \texttt{@media print} (\texttt{InvoiceModal.jsx}).
\end{enumerate}

\end{enumerate}
